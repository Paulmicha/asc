$$
\hat p(\ell, t) \;=\; \frac{k_{\ell,t} + \alpha_t}{n_{\ell,t} + \alpha_t + \beta_t},
\qquad
\frac{\alpha_t}{\alpha_t + \beta_t} = p^{0}_{t},\quad \alpha_t + \beta_t = 4
$$
